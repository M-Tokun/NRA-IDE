# ─────────────────────────────────────────────
# DynamicState (Normalized State Quantity / Container)
# ─────────────────────────────────────────────

@dataclass
class DynamicState:
    """
    Medium-independent normalized state quantity. value ∈ [0.0, 1.0].

    All dynamic calculations pass through this container.
    Differences in medium are expressed by Law (transition functions);
    the container's shape does not change.

    Field Descriptions:
      value   : Current state value (0=fully stable, 1=limit point)
      rate    : Rate of change dv/dt (velocity)
      buffer  : Energy awaiting dissipation (error sink; do not erase)
      tension : Restoring force generated by distance from constraints (tension)
      history : History of instantaneous values (peak records; do not average)
    """
    value  : float = 0.0
    rate   : float = 0.0
    buffer : float = 0.0
    tension: float = 0.0
    history: List[float] = field(default_factory=list)

    # ── [EXT: pressure] ──────────────────────────────
    # Pressure term (human body / fluid systems)
    # Currently: +0.0 (unused)
    # When implementing: Add pressure: float = 0.0,
    #   In TensionLaw.compute():
    #   tension += pressure_coefficient * self.pressure
    pressure: float = 0.0   # [EXT: pressure] +0.0 (unused)

    # ── [EXT: viscosity] ─────────────────────────────
    # Viscosity coefficient (suppresses rate of change; makes abrupt peaks physically impossible)
    # Currently: x1 (multiplication, so disabled)
    # When implementing: rate_limited = rate * viscosity_factor (0 < v_factor <= 1)
    viscosity_factor: float = 1.0   # [EXT: viscosity] x1 (disabled)

    # ── [EXT: plasticity] ────────────────────────────
    # Plasticity deformation coefficient (shifts the reference value after excess; history-dependent)
    # Currently: x1 (no reference value change)
    # When implementing: threshold_shift += plasticity * max(0, value - threshold)
    plasticity_factor: float = 1.0  # [EXT: plasticity] x1 (disabled)

    def record(self, v: float) -> None:
        """Record instantaneous value as is (do not average)"""
        self.history.append(v)

    def peak(self) -> float:
        """Peak value (maximum instantaneous value)"""
        return max(self.history) if self.history else 0.0

    def normalize(self, raw: float, raw_max: float) -> float:
        """Normalize dimensional physical quantity to [0,1]"""
        return min(1.0, max(0.0, raw / raw_max)) if raw_max > 0 else 0.0

    def denormalize(self, norm: float, raw_max: float) -> float:
        """Reverse transform [0,1] to dimensional physical quantity"""
        return norm * raw_max


# ─────────────────────────────────────────────
# EnergyLedger (Energy Balance Ledger)
# ─────────────────────────────────────────────

@dataclass
class EnergyLedger:
    """
    Recorder for energy conservation law.
    Verify E_in = E_stored + E_dissipated (conservation) every cycle.

    Escape errors to buffer (do not erase).
    """
    E_in       : float = 0.0   # Input energy (disturbance / action cost)
    E_stored   : float = 0.0   # Energy stored in the system
    E_dissipated: float = 0.0  # Dissipated energy (released via buffer)

    # ── [EXT: thermal_dissipation] ───────────────────
    # Thermal dissipation (loss during high-speed fluctuations; currently +0.0)
    E_thermal  : float = 0.0   # [EXT: thermal_dissipation] +0.0

    def balance(self) -> float:
        """Conservation error = E_in - (E_stored + E_dissipated)"""
        return self.E_in - (self.E_stored + self.E_dissipated + self.E_thermal)

    def is_conserved(self, tol: float = 1e-6) -> bool:
        return abs(self.balance()) < tol


# ─────────────────────────────────────────────
# Token (Information Unit)
# ─────────────────────────────────────────────

@dataclass
class Token:
    """
    Data packet circulating in the ring.
    state: DynamicState (container)
    meta : System management information
    ledger: Energy balance
    history: Observation log (post-facto records only)
    """
    state  : DynamicState = field(default_factory=DynamicState)
    meta   : Dict[str, Any] = field(default_factory=dict)
    ledger : EnergyLedger = field(default_factory=EnergyLedger)
    history: List[Tuple[str, DynamicState]] = field(default_factory=list, repr=False)

    def __post_init__(self):
        self.meta.setdefault("cycle", 0)
        self.meta.setdefault("created_at", jst_now())
        self.meta.setdefault("violated_laws", [])
        self.meta.setdefault("infeasible", False)

    def snapshot(self, node_name: str) -> None:
        self.history.append((node_name, copy.deepcopy(self.state)))

    def increment_cycle(self) -> None:
        self.meta["cycle"] += 1

    def log_violation(self, law_name: str) -> None:
        self.meta["violated_laws"].append({
            "law": law_name,
            "cycle": self.meta["cycle"],
            "value": self.state.value,
            "tension": self.state.tension,
        })

    def __repr__(self) -> str:
        s = self.state
        return (f"Token(cycle={self.meta['cycle']}, "
                f"value={s.value:.3f}, rate={s.rate:.3f}, "
                f"buffer={s.buffer:.3f}, tension={s.tension:.3f})")


# ─────────────────────────────────────────────
# Law (Constraint / Dynamic Rule)
# ─────────────────────────────────────────────

class Law(ABC):
    """
    Abstract base for dynamic rules acting on DynamicState.

    check()  : Whether the constraint is satisfied
    tension(): Restoring force proportional to distance from constraint (tension)
    correct(): Correct state using tension. Escape error to buffer.
    """

    @property
    def name(self) -> str:
        return self.__class__.__name__

    @abstractmethod
    def check(self, token: Token) -> bool: ...

    @abstractmethod
    def tension(self, token: Token) -> float:
        """
        Calculate restoring force.
        Distance (excess) generates force—the principle of not deriving force from distance
        here means "not treating distance itself as force".
        Transformation via proportional coefficient k is allowed.
        """
        ...

    @abstractmethod
    def correct(self, token: Token) -> Optional[Token]:
        """
        Correct using tension. Escape excess energy to buffer (do not erase).
        If uncorrectable → Return None (infeasibility).
        """
        ...


class BoundLaw(Law):
    """
    General upper bound constraint.
    Ensures DynamicState.value does not exceed threshold.

    Tension: T = k * max(0, value - threshold)
    Correction: Pull value back to threshold - margin, escape excess to buffer
    """

    def __init__(
        self,
        threshold: float = 0.8,    # Upper limit (normalized value)
        k        : float = 1.0,    # Tension coefficient
        margin   : float = 0.05,   # Clip margin (safety zone for next cycle)
    ):
        self.threshold = threshold
        self.k         = k
        self.margin    = margin

        # ── [EXT: viscosity_coupling] ─────────────────
        # Couple viscosity to tension calculation (add resistance if rate of change is large)
        # Currently: +0.0
        # When implementing: T += viscosity * abs(token.state.rate)
        self._viscosity_coupling: float = 0.0  # [EXT: viscosity_coupling] +0.0

        # ── [EXT: brittle_threshold] ──────────────────
        # Brittle fracture point (abrupt tension release if this value is exceeded)
        # Currently: x1 (remains normal linear tension)
        # When implementing: if value > brittle_threshold: T *= brittle_amplifier
        self._brittle_factor: float = 1.0      # [EXT: brittle_threshold] x1

    def check(self, token: Token) -> bool:
        return token.state.value <= self.threshold

    def tension(self, token: Token) -> float:
        excess = max(0.0, token.state.value - self.threshold)
        T = self.k * excess
        T += self._viscosity_coupling          # [EXT: viscosity_coupling]
        T *= self._brittle_factor              # [EXT: brittle_threshold]
        return T

    def correct(self, token: Token) -> Optional[Token]:
        T = self.tension(token)
        if T <= 0:
            return token

        excess = token.state.value - self.threshold
        target = self.threshold - self.margin

        # Escape excess energy to buffer (do not erase)
        token.state.buffer += excess
        token.state.value   = max(0.0, target)
        token.state.tension = T

        # FIX: Accumulate differential to E_stored (instead of overwrite)
        token.ledger.E_stored += excess

        token.log_violation(self.name)
        return token


class RateLimitLaw(Law):
    """
    Rate of change upper limit constraint.
    Suppresses abrupt changes (chattering prevention).

    Tension: T = k * max(0, rate - max_rate)
    Correction: Clamp rate to max_rate - margin, escape excess to buffer
    """

    def __init__(
        self,
        max_rate: float = 0.2,     # Maximum allowed rate (normalized)
        k       : float = 1.0,    # Tension coefficient
        margin  : float = 0.01,   # Safety margin
    ):
        self.max_rate = max_rate
        self.k        = k
        self.margin   = margin

    def check(self, token: Token) -> bool:
        return abs(token.state.rate) <= self.max_rate

    def tension(self, token: Token) -> float:
        excess = max(0.0, abs(token.state.rate) - self.max_rate)
        return self.k * excess

    def correct(self, token: Token) -> Optional[Token]:
        T = self.tension(token)
        if T <= 0:
            return token

        sign = math.copysign(1, token.state.rate)
        target_rate = sign * (self.max_rate - self.margin)
        excess = abs(token.state.rate) - self.max_rate

        token.state.buffer += excess
        token.state.rate    = target_rate
        token.state.tension = T

        # Accumulate to E_stored
        token.ledger.E_stored += excess

        token.log_violation(self.name)
        return token


# ─────────────────────────────────────────────
# Node (Processing Unit)
# ─────────────────────────────────────────────

class Node(ABC):
    """
    Abstract base for ring nodes.
    process(): Token processing
    _record() : Post-process record (observation)
    """

    def __init__(self, name: str):
        self.name = name
        self._logger = logging.getLogger(f"NRA.{name}")

    @abstractmethod
    def process(self, token: Token) -> Token: ...

    def _record(self, token: Token) -> Token:
        token.snapshot(self.name)
        return token


class SafetyNode(Node):
    """
    Inference diode and safety verification node.
    Apply all Laws.
    """

    def __init__(self, name: str, laws: List[Law]):
        super().__init__(name)
        self.laws = laws

    def process(self, token: Token) -> Token:
        for law in self.laws:
            if not law.check(token):
                result = law.correct(token)
                if result is None:
                    # Uncorrectable → Fail-Closed (silence)
                    token.meta["infeasible"] = True
                    return self._record(token)
                token = result
        return self._record(token)


class DynamicsNode(Node):
    """
    State transition node.
    Transition function can be injected externally.
    Replace for each medium.
    """

    def __init__(
        self,
        name: str,
        transition_fn: Callable[[DynamicState, int], DynamicState] = None,
    ):
        super().__init__(name)
        self.transition_fn = transition_fn or self._default_transition

    @staticmethod
    def _default_transition(state: DynamicState, cycle: int) -> DynamicState:
        """
        Default transition function. Replace to use.
        """
        cost = 0.1   # Base cost

        # ── [EXT: viscosity] ─────────────────────────
        # Viscosity term: Suppress rate change
        # Currently: x1 (no suppression)
        # When implementing: cost *= state.viscosity_factor
        cost *= state.viscosity_factor   # [EXT: viscosity] x1

        # ── [EXT: pressure] ──────────────────────────
        # Pressure term: Additional cost
        # Currently: +0.0
        # When implementing: cost += state.pressure * pressure_coefficient
        cost += state.pressure * 0.0     # [EXT: pressure] +0.0

        new_rate  = cost
        new_value = min(1.0, state.value + cost)
        state.rate  = new_rate
        state.value = new_value
        state.record(new_value)          # Record instantaneous value (do not average)
        return state

    def process(self, token: Token) -> Token:
        if token.meta.get("infeasible"):
            return self._record(token)
        token.state = self.transition_fn(token.state, token.meta["cycle"])
        # Energy balance: Record input amount
        token.ledger.E_in += token.state.rate
        return self._record(token)


class DisturbanceNode(Node):
    """
    Disturbance injection node.
    Specified in normalized value (magnitude ∈ [0, 1]).
    """

    def __init__(
        self,
        name: str,
        trigger_cycles: List[int],
        magnitude: float = 0.5,
    ):
        super().__init__(name)
        self.trigger_cycles = trigger_cycles
        self.magnitude = magnitude

    def process(self, token: Token) -> Token:
        if token.meta["cycle"] in self.trigger_cycles:
            self._logger.warning(
                f"Disturbance injection [cycle={token.meta['cycle']}]: +{self.magnitude}"
            )
            prev = token.state.value
            token.state.value = min(1.0, token.state.value + self.magnitude)
            token.state.rate   = token.state.value - prev
            token.ledger.E_in += self.magnitude   # Account in energy balance
            token.state.record(token.state.value)
        return self._record(token)


# ─────────────────────────────────────────────
# NomologicalRing (Nomological Ring Engine)
# ─────────────────────────────────────────────

class NomologicalRing:
    def __init__(self, nodes: List[Node]):
        self.nodes = nodes
        self._logger = logging.getLogger("NRA.Ring")

    def run_cycle(self, token: Token) -> Token:
        self._logger.debug(f"=== Cycle {token.meta['cycle']} ===")
        for node in self.nodes:
            token = node.process(token)
            if token.meta.get("infeasible"):
                self._logger.warning(f"Infeasibility @ {node.name}")
                break
        token.increment_cycle()
        return token

    def run(self, token: Token, max_cycles: int = 20) -> Token:
        for _ in range(max_cycles):
            token = self.run_cycle(token)
        return token

class ObservationLog:
    """Pure observation/visualization layer that does not interfere with processing"""

    def __init__(self):
        self.records: List[Dict[str, Any]] = []

    def record(self, token: Token) -> None:
        s = token.state
        self.records.append({
            "cycle"    : token.meta["cycle"],
            "value"    : s.value,
            "rate"     : s.rate,
            "buffer"   : s.buffer,
            "tension"  : s.tension,
            "peak"     : s.peak(),
            "E_in"     : token.ledger.E_in,
            "E_stored" : token.ledger.E_stored,
            "E_dissip" : token.ledger.E_dissipated,
            "conserved": token.ledger.is_conserved(),
            # FIX: Detect all violations in the same cycle (fixed issue of referencing only [-1])
            "violated" : any(
                v["cycle"] == token.meta["cycle"] - 1
                for v in token.meta["violated_laws"]
            ),
        })

    def print_table(self) -> None:
        print(f"\n{'cycle':>5} | {'value':>6} | {'rate':>6} | "
              f"{'buffer':>7} | {'tension':>7} | {'peak':>6} | "
              f"{'E_bal':>6} | status")
        print("-" * 72)
        for r in self.records:
            bal = r["E_in"] - r["E_stored"] - r["E_dissip"]
            viol = "⚠️ " if r["violated"] else "✅"
            print(
                f"{r['cycle']:5d} | {r['value']:6.3f} | {r['rate']:+6.3f} | "
                f"{r['buffer']:7.3f} | {r['tension']:7.4f} | {r['peak']:6.3f} | "
                f"{bal:+6.4f} | {viol}"
            )

    def plot(self, save_path: str, threshold: float = 0.8) -> None:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt

        cycles  = [r["cycle"] for r in self.records]
        values  = [r["value"]  for r in self.records]
        buffers = [r["buffer"] for r in self.records]
        tension = [r["tension"] for r in self.records]

        fig, axes = plt.subplots(3, 1, figsize=(13, 10), sharex=True)

        # ── Upper: state.value vs threshold ──
        axes[0].plot(cycles, values, marker="o", color="steelblue",
                     linewidth=2, label="state.value")
        axes[0].axhline(y=threshold, color="crimson", linestyle="--",
                        linewidth=1.5, label=f"threshold={threshold}")
        axes[0].fill_between(cycles, 0, threshold, color="limegreen",
                             alpha=0.07, label="Safe zone")
        for r in self.records:
            if r["violated"]:
                axes[0].axvspan(r["cycle"]-0.4, r["cycle"]+0.4,
                                color="orange", alpha=0.2)
        axes[0].set_ylabel("Normalized Value")
        axes[0].set_title("NRA-IDE Foundation: DynamicState Dynamics")
        axes[0].legend(fontsize=8)
        axes[0].grid(True, alpha=0.3)

        # ── Middle: buffer (error sink) and tension ──
        axes[1].fill_between(cycles, 0, buffers, color="dodgerblue",
                             alpha=0.4, label="buffer (dissipated energy)")
        axes[1].plot(cycles, tension, color="crimson", linewidth=1.5,
                     label="tension (restoring force)")
        axes[1].set_ylabel("Buffer / Tension")
        axes[1].legend(fontsize=8)
        axes[1].grid(True, alpha=0.3)

        # ── Lower: Energy balance ──
        e_in    = [r["E_in"]    for r in self.records]
        e_store = [r["E_stored"] for r in self.records]
        e_diss  = [r["E_dissip"] for r in self.records]
        axes[2].plot(cycles, e_in,    color="gold",      linewidth=1.5, label="E_in")
        axes[2].plot(cycles, e_store, color="steelblue", linewidth=1.5, label="E_stored")
        axes[2].plot(cycles, e_diss,  color="tomato",    linewidth=1.5, label="E_dissipated")
        axes[2].set_xlabel("Cycle")
        axes[2].set_ylabel("Energy")
        axes[2].legend(fontsize=8)
        axes[2].grid(True, alpha=0.3)
        axes[2].set_xticks(cycles)

        plt.tight_layout()
        plt.savefig(save_path, dpi=150)
        print(f"\n📊 Plot saved: {save_path}")
        plt.close()
