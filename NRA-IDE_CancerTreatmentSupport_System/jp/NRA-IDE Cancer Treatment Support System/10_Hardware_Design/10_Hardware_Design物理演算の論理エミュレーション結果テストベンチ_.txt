計算開始 (T+1): calc_trigger が立ち上がり、演算パイプラインの Stage 1 が開始されます。
パイプライン完了 (T+4): 3クロック後、o_out_valid が立ち上がります。

# Phase 10 Logic Verification Script
# 26-0203-1650 JST

def verify_nra_logic():
    # 入力値 (Q8.8 固定小数点)
    stiff = 0x0180; visc = 0x000D; diam = 0x0C00
    pore = 0x0800; flow = 0x0099; boost = 0x0000; vel = 0x3200

    # Stage 1: Geometry
    dx = (diam - pore) if diam >= pore else 0

    # Stage 2: Multipliers (Result is Q16.16)
    f_elastic = (stiff + boost) * dx
    f_viscous = visc * vel

    # Stage 3: Comparison
    total_resist = f_elastic + f_viscous
    flow_force = flow << 8 # Q8.8 to Q16.16 scaling

    is_jammed = 1 if total_resist > flow_force else 0

    print(f"Total Resistance (Q16.16): {total_resist}")
    print(f"Flow Force (Q16.16):       {flow_force}")
    print(f"Verification Result:       {'JAMMED' if is_jammed else 'SAFE'}")

verify_nra_logic()
